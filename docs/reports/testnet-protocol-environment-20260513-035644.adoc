[source,bash]
----
stellar contract invoke --network testnet --source-account <user-key> --id "CBR6GCCBB73UMQOHUXWFS4L5BL7X7GUR3EPM5YF4OWU6LRNBTARPQWTR" -- deposit --amount 1000000000 --from <G...USER> --asset "CD37EBBWP3RY4QRIBP4D7KJGN6GEFXNHF5ZDMXSKSHVTLJMZ3CSIQMP7" --min-shares-out 0

----
